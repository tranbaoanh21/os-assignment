\section*{2. Kiến trúc và vòng đời process}

\begin{tabularx}{\textwidth}{>{\ttfamily\color{navy}}p{3.2cm} X}
\toprule
\textbf{File} & \textbf{Vai trò}\\
\midrule
src/os.c & Khởi tạo simulator, tạo CPU/loader thread và điều khiển vòng đời process.\\
src/loader.c & Đọc process description, tạo PCB và parse instruction.\\
src/cpu.c & Lấy instruction tại PC, tăng PC và dispatch sang memory hoặc syscall.\\
include/common.h & Định nghĩa PCB, kernel structure, instruction và registers.\\
\bottomrule
\end{tabularx}

\flow{Process file -> load() -> PCB -> add\_proc() -> get\_proc() -> run()}

\textbf{Hàm cần nhớ:} \func{load()}, \func{run()}, \func{add\_proc()},
\func{get\_proc()}, \func{put\_proc()}, \func{finish\_proc()}.

\section*{3. Scheduler, multi-CPU và timer}

\begin{tabularx}{\textwidth}{>{\ttfamily\color{navy}}p{3.2cm} X}
\toprule
\textbf{File} & \textbf{Vai trò}\\
\midrule
src/sched.c & 140 MLQ ready queues, weighted quota, running list và queue lock.\\
src/queue.c & FIFO operations: enqueue, dequeue và purge.\\
src/timer.c & Barrier đồng bộ các simulated device theo từng time slot.\\
src/os.c & CPU thread gọi scheduler và trả PCB khi hết quantum.\\
\bottomrule
\end{tabularx}

\flow{READY -> get\_mlq\_proc() -> RUNNING -> put\_mlq\_proc() / FINISHED}

\begin{itemize}
  \item \func{get\_mlq\_proc()}: quét priority, giảm slot và đưa PCB vào running list.
  \item \func{put\_mlq\_proc()}: hết quantum, đưa PCB từ running về ready queue.
  \item \func{next\_slot()}: device báo hoàn tất slot rồi chờ timer giải phóng.
  \item \func{queue\_lock}: bảo đảm hai CPU không lấy cùng một PCB.
\end{itemize}

\section*{4. Memory, paging và region reuse}

\begin{tabularx}{\textwidth}{>{\ttfamily\color{navy}}p{3.2cm} X}
\toprule
\textbf{File} & \textbf{Vai trò}\\
\midrule
src/libmem.c & Memory API, logical region allocation/free và copy operations.\\
src/mm-vm.c & VMA, tăng sbrk và các thao tác virtual-memory mapping.\\
src/mm-memphy.c & Mô phỏng RAM/SWAP; cấp, đọc, ghi và trả physical frame.\\
src/sys\_mem.c & Kernel handler thực thi memory operations sau khi kiểm tra caller.\\
include/os-mm.h & Các cấu trúc mm, VMA, region và page-table metadata.\\
\bottomrule
\end{tabularx}

\flow{cpu.c -> liballoc() -> syscall 17 -> \_\_sys\_memmap() -> \_\_alloc()}

\begin{itemize}
  \item \func{\_\_alloc()}: ưu tiên lấy vùng từ free-region list trước khi tăng \func{sbrk}.
  \item \func{\_\_free()}: trả logical region về free-region list.
  \item \func{MEMPHY\_get\_freefp()}: lấy physical frame trống.
  \item \file{src/mem.c}: legacy memory model khi không bật \texttt{MM\_PAGING}.
\end{itemize}

\section*{6. Multi-level paging 64-bit và canonical address}

\begin{tabularx}{\textwidth}{>{\ttfamily\color{navy}}p{3.2cm} X}
\toprule
\textbf{File} & \textbf{Vai trò}\\
\midrule
include/mm64.h & Bit ranges, masks và macro lấy index PGD/P4D/PUD/PMD.\\
src/mm64.c & Canonical validation, sparse page tables và five-level walk.\\
include/os-mm.h & Lưu page-table root và thống kê walks/accesses/bytes.\\
src/mm-memphy.c & Cấp frame vật lý cho các mapping mới.\\
\bottomrule
\end{tabularx}

\flow{canonical check -> PGD -> P4D -> PUD -> PMD -> PT -> frame}

\begin{itemize}
  \item \func{paging64\_is\_canonical()}: từ chối địa chỉ 64-bit không hợp lệ.
  \item \func{walk\_pte()}: đi qua năm cấp và tùy chọn cấp sparse branch.
  \item \func{vmap\_pgd\_memset()}: map page tại virtual address.
  \item \func{paging64\_owns\_fpn()}: kiểm tra process có sở hữu frame không.
\end{itemize}

\file{include/mm64.h} $\rightarrow$ \file{src/mm64.c} $\rightarrow$ \file{include/os-mm.h}\\
\bottomrule
\end{tabularx}
